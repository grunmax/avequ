\begin{abstract}
This document collects the core formulae of the Lyapunov proof for V.F.S. v2.0 (Open-Gate) and contains no graphs. It records the base dynamics, normalized variables, Lyapunov functional, active-domain walls, radial corridor, bounded Open-Gate source, memory bounds, non-Zeno estimate, and asymptotic cleansing statements. In addition, the central dissipative estimate $\dot\LV\le C-C_1\LV$ --- previously stated as a boxed result --- is here \emph{derived term by term}: the contraction rate $C_1$ is built from exactly two sign-definite blocks (resistance cleansing and will--action alignment), while the Sophia, radial and Open-Gate contributions are bounded and enter only the constant $C$. The auxiliary light-filter constant is proved positive in closed form, so that $C_1>0$ holds for \emph{every} admissible parameter set under the standing walls, not merely for the numerical witness.
\end{abstract}

\section{Radial Corridor}

Radial identity from $r^2=h_1h_2+e(h_3)$:
\begin{equation}
  2r\dot r=
  \mu\bigl[a(h_1^2+h_2^2)+cq(h_1+h_2)\bigr]
  -2\rho(r^2-e(h_3))-2\alpha q r^2
  =:\Rcal_{\rm rad}.
\end{equation}

Maximum $e$ on the active corridor:
\begin{equation}
  e_*^{\max}=\varepsilon\frac{(1-h_{3,*})^2}{\Lambda_c^2}.
\end{equation}

Non-degeneracy:
\begin{equation}
  r_*^2>e_*^{\max}.
\end{equation}

Active margin:
\begin{equation}
  r_*+h_{3,*}-1\ge \eta\frac{1-h_{3,*}}{\Lambda_c}.
\end{equation}

Lower radial wall:
\begin{equation}
  a(1-r_*)\bigl(r_*^2-e_*^{\max}\bigr)
  \ge (\rho+\alpha q^*)r_*^2.
\end{equation}

This wall is sufficient, not tight. The exact lower-wall requirement is
\begin{equation}
  (1-r_*)a(r_*^2-e)
  \ge
  \rho(r_*^2-e)+\alpha q r_*^2,
\end{equation}
and the stated form replaces \(\rho(r_*^2-e)\) by the larger \(\rho r_*^2\).

Upper radial wall:
\begin{equation}
  (1-r^*)\bigl[a(d_*^2+2r^{*2})+cq^*\sqrt{d_*^2+4r^{*2}}\bigr]
  \le 2\rho\bigl(r^{*2}-e_*^{\max}\bigr).
\end{equation}

Direct upper radial inwardness condition:
\begin{equation}
  \mu\left[a(h_1^2+h_2^2)+cq(h_1+h_2)\right]
  \le 2\rho(r^{*2}-e)+2\alpha q r^{*2}
  \qquad\text{on } r=r^*.
\end{equation}

Open-Gate radial convention:
\begin{equation}
  q_{\rm new}^*:=\sqrt{\frac{\delta r^*+S_q^{\max}}{\alpha}}.
\end{equation}

Conservative lower radial wall with $q_{\rm new}^*$:
\begin{equation}
  a(1-r_*)\bigl(r_*^2-e_*^{\max}\bigr)
  \ge (\rho+\alpha q_{\rm new}^*)r_*^2.
\end{equation}

Epektasis-dominance derivative at the upper radial wall:
\begin{equation}
  \frac{\partial\Rrad}{\partial q}\bigg|_{r^*}

\end{equation}

Worst-case bound:
\begin{equation}
  \frac{\partial\Rrad}{\partial q}\bigg|_{r^*}
  \le (1-r^*)c\sqrt{d_*^2+4r^{*2}}-2\alpha r^{*2}.
\end{equation}

Epektasis-dominance sufficient condition:
\begin{equation}
  \boxed{
  2\alpha r^{*2}\ge (1-r^*)c\sqrt{d_*^2+4r^{*2}}

\section{Lyapunov Dissipative Estimate}

This section derives the estimate $\dot\LV\le C-C_1\LV$ term by term and exhibits
an explicit, strictly positive $C_1$.

\subsection{Standing admissibility data}
We use the conditions that already define and protect the active domain $\DA$:
\begin{equation}
  \text{(A1)}\ a>0,\ \delta>0,\ \gamma>0;
  \qquad
  \text{(A2)}\ 0\le h_{3,*}<1,\ 0<r_*<r^*<1,\ \mu_*:=1-r^*>0;
\end{equation}
\begin{equation}
  \text{(A3)}\ m_*:=r_*+h_{3,*}-1>0\ \ (\text{active margin});
  \qquad
  \text{(A4)}\ \kappa>0,\ 0<C_\sigma<\infty.
\end{equation}
Inside $\DA$: $\sigma\in[0,C_\sigma]$, $q\in[0,q^*]$, $h_3\in[h_{3,*},1)$,
$r\in[r_*,r^*]$, $D_\Delta\le D^*$, and $0\le \Sq\le \Sq^{\max}$. Write
\begin{equation}
  T:=\tanh(\kappa\sigma)\in[0,1],
  \qquad
  \Om=\frac{\Lambda_c}{1-h_3},
  \qquad
  m:=r+h_3-1=\frac{u-\Lambda_c}{\Om}\ge m_*>0,
\end{equation}
so that the resistance equation reads $\dot\sigma=(\gamma-\delta u)T=-\delta\Om m\,T$.

\subsection{Derivative decomposition}
With $\LV=A_\sigma\Psis(\sigma)+A_\Delta D_\Delta+A_qq^2+A_rB(r)$,
$B(r)=-\ln(1-r)$, and $\Psis'(\sigma)=T$, $B'(r)=1/(1-r)$,
\begin{equation}
  \dot\LV=
  \underbrace{A_\sigma T\dot\sigma}_{\text{(I)}}
  +\underbrace{A_\Delta\dot D_\Delta}_{\text{(II)}}
  +\underbrace{2A_qq\dot q}_{\text{(III)}}
  +\underbrace{A_r\frac{\dot r}{1-r}}_{\text{(IV)}}.
\end{equation}

\subsection{Block (I): resistance --- contracting}
Substituting $\dot\sigma=-\delta\Om m\,T$,
\begin{equation}
  \text{(I)}=-A_\sigma\,\delta\,\Om\,m\,T^2\le0.
\end{equation}
By Lemma~\ref{lem:cpsi} below $T^2\ge c_\Psi\Psis$ on $[0,C_\sigma]$, and
$\Om\ge\Omega_{\min}:=\Lambda_c/(1-h_{3,*})$, $m\ge m_*$. Hence
\begin{equation}
  \boxed{\ \text{(I)}\le -\kappa_\sigma A_\sigma\Psis,
  \qquad
  \kappa_\sigma:=\delta\,\Omega_{\min}\,m_*\,c_\Psi.\ }
\end{equation}

\subsection{Block (II): balance --- contracting}
From $\dot D_\Delta=-2(a\mu+\rho+\alpha q)D_\Delta$ and
$a\mu+\rho+\alpha q\ge a\mu_*$,
\begin{equation}
  \boxed{\ \text{(II)}\le -\kappa_\Delta A_\Delta D_\Delta,
  \qquad
  \kappa_\Delta:=2a\mu_*.\ }
\end{equation}

\subsection{Block (III): Sophia --- bounded}
Expanding the normalized $q$-dynamics $\dot q=\delta mT-\alpha q^2+\Sq$,
\begin{equation}
  \frac{d}{dt}(A_qq^2)=2A_qq\dot q
  =-2A_q\alpha q^3
  +\underbrace{2A_q\,\delta mT\,q}_{\text{transfer}}
  +\underbrace{2A_q\Sq\,q}_{\text{gate}}.
\end{equation}
The transfer term shares the factor $\delta mT$ with block~(I): the conversion
that \emph{drains} $\sigma$ \emph{feeds} $q$. Both positive terms are bounded on
$\DA$,
\begin{equation}
  0\le 2A_q\delta mTq\le 2A_q\delta m^{\max}q^*,
  \qquad
  0\le 2A_q\Sq q\le 2A_qq^*\Sq^{\max}=:\Delta C_{\rm gate},
\end{equation}
while $-2A_q\alpha q^3\le0$. Since $C_1q^2-2A_q\alpha q^3$ is bounded above on the
compact $[0,q^*]$ for any $C_1$, block~(III) obeys
\begin{equation}
  \text{(III)}\le -C_1A_qq^2+B_q,
  \qquad
  B_q<\infty;
\end{equation}
it contributes only to the constant.

\subsection{Block (IV): radial --- bounded}
$B(r)=-\ln(1-r)$ diverges only as $r\to1^-$, but $\DA$ caps $r\le r^*<1$, so
$B(r)\le-\ln(1-r^*)<\infty$. With $r\ge r_*>0$, $\mu\ge\mu_*>0$ and bounded
$h_1,h_2$, the radial rate $\dot r=\mathcal R_{\rm rad}/(2r)$ is bounded, so
block~(IV) is bounded on $\DA$ and contributes only to the constant. Its sign on
the face $r=r^*$ (i.e.\ $\mathcal R_{\rm rad}\le0$, the upper radial wall) is used
for invariance, not for $C_1$.

\subsection{The light-filter constant}
\begin{lemma}\label{lem:cpsi}
For every $\kappa>0$ and $C_\sigma<\infty$,
\begin{equation}
  c_\Psi:=\min_{0\le\sigma\le C_\sigma}
  \frac{\tanh^2(\kappa\sigma)}{\Psis(\sigma)}
  =\kappa\,\frac{\tanh^2(\kappa C_\sigma)}{\ln\cosh(\kappa C_\sigma)}>0,
  \qquad
  \Psis(\sigma)=\tfrac1\kappa\ln\cosh(\kappa\sigma).
\end{equation}
\end{lemma}
\begin{proof}
With $t=\kappa\sigma$ the ratio equals $\kappa\,g(t)$,
$g(t)=\tanh^2t/\ln\cosh t$ on $(0,\kappa C_\sigma]$. As $t\to0^+$,
$\tanh^2t\sim t^2$ and $\ln\cosh t\sim t^2/2$, so $g(t)\to2$ and $g$ extends
continuously to $g(0)=2$; moreover $g>0$ on $(0,\infty)$ since
$\tanh^2t>0$, $\ln\cosh t>0$. The sign of $g'$ equals that of
$h(t):=2\sech^2t\,\ln\cosh t-\tanh^2t$. Now $h(0)=0$ and
\begin{equation}
  h'(t)=-4\,\sech^2t\,\tanh t\,\ln\cosh t<0\quad(t>0),
\end{equation}
so $h(t)<0$ for $t>0$, whence $g'<0$ and $g$ is strictly decreasing on
$(0,\infty)$. The minimum over $[0,\kappa C_\sigma]$ is thus attained at the
right endpoint, giving the closed form, which is positive.
\end{proof}

\subsection{Dissipative estimate}
\begin{proposition}[Dissipative estimate with explicit $C_1$]\label{prop:diss}
Under \emph{(A1)--(A4)} there exist $C_1>0$ and $C<\infty$ with
\begin{equation}
  \boxed{\ \dot\LV\le C-C_1\LV
  \quad\text{on }\DA,
  \qquad
  C_1=\min\{\kappa_\sigma,\kappa_\Delta\}
  =\min\bigl\{\delta\,\Omega_{\min}\,m_*\,c_\Psi,\ 2a\mu_*\bigr\}>0.\ }
\end{equation}
The Open-Gate source affects only $C$ through
$0\le\Delta C_{\rm gate}\le 2A_qq^*\Sq^{\max}$ and leaves $C_1$ unchanged.
\end{proposition}
\begin{proof}
Summing (I)--(IV),
\begin{equation}
  \dot\LV\le-\kappa_\sigma A_\sigma\Psis-\kappa_\Delta A_\Delta D_\Delta+C',
\end{equation}
with $C'<\infty$ collecting the bounded blocks (III)--(IV) (transfer, gate,
radial). Set $C_1=\min\{\kappa_\sigma,\kappa_\Delta\}$, so that
$-\kappa_\sigma A_\sigma\Psis-\kappa_\Delta A_\Delta D_\Delta\le
-C_1(A_\sigma\Psis+A_\Delta D_\Delta)$. Since $A_qq^2+A_rB(r)\le P_{\max}<\infty$
on $\DA$,
\begin{equation}
  \dot\LV\le -C_1\LV+C_1P_{\max}+C'=:-C_1\LV+C.
\end{equation}
\emph{Finiteness of $C$.} Blocks (III),(IV) are bounded; the only block
unbounded as $h_3\to1^-$ ($\Om\to\infty$) is (I)$=-A_\sigma\delta\Om mT^2\le0$;
hence $\sup_{\DA}(\dot\LV+C_1\LV)<\infty$.
\emph{Positivity of $C_1$.} $\kappa_\Delta=2a\mu_*>0$ by (A1),(A2); and
$\kappa_\sigma=\delta\,\Omega_{\min}\,m_*\,c_\Psi>0$ because $\delta>0$ (A1),
$\Omega_{\min}=\Lambda_c/(1-h_{3,*})>0$ (A2), $m_*>0$ (A3) and $c_\Psi>0$
(Lemma~\ref{lem:cpsi} with (A4)). Each factor is one of the standing
admissibility data, so $C_1>0$ for every admissible parameter set.
\end{proof}

Open-Gate constant split:
\begin{equation}
  C=C_{\rm closed}+\Delta C_{\rm gate},
  \qquad
  0\le\Delta C_{\rm gate}\le 2A_q q^*\Sq^{\max}.
\end{equation}

\textbf{Key structural point.}
The dissipation rate $C_1$ is produced solely by the two sign-definite
contracting modes --- resistance cleansing (I) at rate $\kappa_\sigma$ and
will--action alignment (II) at rate $\kappa_\Delta$. Sophia/Epektasis ($q^2$),
radial position ($B(r)$) and received grace ($\Sq$) cannot accelerate the decay
of $\LV$; they only enlarge the bounded constant $C$. Thus $C_1$ is unchanged by
the Open-Gate source, while $C$ receives only the bounded contribution
$\Delta C_{\rm gate}$.

\begin{remark}[No uniform floor at the admissibility boundary]
Positivity is per parameter set, not uniform: $\inf C_1=0$ over the whole
admissible set, approached as $m_*\to0^+$ (a set on the active-margin wall) or
$\mu_*\to0^+$ ($r^*\to1$). A configuration pinned on the active margin has zero
cleansing rate. The correct statement is ``$C_1>0$ for each admissible set,''
not a single $C_1>0$ valid for all sets at once.
\end{remark}

\begin{remark}[Relation to the numerical witness]
At the witness point of Section~\ref{sec:witness} (with $\kappa=1$, $C_\sigma=2$,
unit weights) one gets $c_\Psi=0.701$, $\kappa_\sigma=0.255$,
$\kappa_\Delta=0.755$, hence $C_1=0.255$ (resistance-limited), versus an
empirical envelope slope $\approx1.5$; the analytic rate is the conservative
guaranteed value, as expected.
\end{remark}

\section{Reset Admissibility (Resurrectio Jump)}

The resets $\reset$ are \emph{exogenous}: since $\DA$ is positively invariant, the
continuous flow never reaches the boundary of the active domain by itself. A death-like
strike is therefore modeled as occurring at an interior point
\begin{equation}
  x^-=(\sigma^-,\lambda^-,V,F,\Om^-)\in\DA
\end{equation}
at an unpredictable time.

\textbf{Reset map (linear metabolism).} With $0\le q_R<1$ and $\kappa_R>0$,
\begin{equation}
  \reset:\quad
  \sigma^+=q_R\sigma^-,\qquad
  \lambda^+=\lambda^-+(1-q_R)\sigma^-,\qquad
  (V,F)\ \text{continuous},\qquad
  \Om^+=\Om^-+\kappa_R\Grec^-.
\end{equation}
Here
\begin{equation}
  \Grec^-:=\Grec(t_{\rm death}^-),
  \qquad
  u^+=\sqrt{VF+\varepsilon}=u^-.
\end{equation}

\textbf{Quantities.}
\begin{equation}
  A_R:=\Grec^-,\qquad
  R_c:=C_\sigma-\sigma_0-\lambda_0,
\end{equation}
\begin{equation}
  R_{\min}^{\rm ceil}:=
  \frac{(1-q_R)\sigma^-}{q^*\kappa_R},
  \qquad
  R_{\max}:=
  \frac{\Om^-(r^--r_*)}{\kappa_R r_*}.
\end{equation}
By the Open-Gate balance
\begin{equation}
  \sigma^-+\lambda^-=
  \sigma_0+\lambda_0+\Grec^-,
\end{equation}
one has
\begin{equation}
  \Grec^-\ge R_c
  \quad\Longleftrightarrow\quad
  \sigma^-+\lambda^-\ge C_\sigma
  \quad\Longleftrightarrow\quad
  \lambda^-\ge C_\sigma-\sigma^-.
\end{equation}

\textbf{Lemma (reset admissibility).}
Let $\kappa_R>0$ and $x^-\in\DA$. If
\begin{equation}
  \max\{R_c,R_{\min}^{\rm ceil}\}\le\Grec^-\le R_{\max},
\end{equation}
then $x^+:=\reset(x^-)\in\DA$ and
\begin{equation}
  \LV(x^+)\le L_R:=\sup_{\DA}\LV<\infty.
\end{equation}

\textbf{Proof.} Coordinate by coordinate,
\begin{equation}
  \sigma^+=q_R\sigma^-\in[0,C_\sigma).
\end{equation}
Moreover,
\begin{equation}
  h_3^+=1-\frac{\Lambda_c}{\Om^+}
  \ge h_3^-\ge h_{3,*},
  \qquad
  h_3^+<1,
\end{equation}
since $\Om^+\ge\Om^-$. Also
\begin{equation}
  h_1^+=h_1^-\frac{\Om^-}{\Om^+},
  \qquad
  h_2^+=h_2^-\frac{\Om^-}{\Om^+},
\end{equation}
so
\begin{equation}
  D_\Delta^+=\frac12\left(d^-\frac{\Om^-}{\Om^+}\right)^2
  \le D_\Delta^-\le D^*,
  \qquad
  h_1^+,h_2^+>0.
\end{equation}
The radial coordinate satisfies
\begin{equation}
  r^+=\frac{r^-\Om^-}{\Om^-+\kappa_R\Grec^-}
  \le r^-\le r^*.
\end{equation}
The lower radial wall is preserved exactly when
\begin{equation}
  r^+\ge r_*
  \quad\Longleftrightarrow\quad
  \kappa_R\Grec^-\le\frac{\Om^-(r^--r_*)}{r_*}
  \quad\Longleftrightarrow\quad
  \Grec^-\le R_{\max}.
\end{equation}
Finally,
\begin{equation}
  q^+=
  \frac{\lambda^-+(1-q_R)\sigma^-}{\Om^-+\kappa_R\Grec^-}
  \ge0.
\end{equation}
Since $x^-\in\DA$ gives $\lambda^-\le q^*\Om^-$,
\begin{equation}
\begin{aligned}
  q^*(\Om^-+\kappa_R\Grec^-)
  -\bigl(\lambda^-+(1-q_R)\sigma^-\bigr)
  &\ge
  q^*\kappa_R\Grec^--(1-q_R)\sigma^-\\
  &\ge0
\end{aligned}
\end{equation}
whenever $\Grec^-\ge R_{\min}^{\rm ceil}$. Hence $q^+\le q^*$.
Thus $x^+\in\DA$. The functional $\LV$ is bounded on $\DA$ because
\begin{equation}
  \Psi_\sigma\le\Psi_\sigma(C_\sigma),
  \qquad
  D_\Delta\le D^*,
  \qquad
  q^2\le(q^*)^2,
  \qquad
  B(r)\le-\ln(1-r^*)<\infty.
\end{equation}
Therefore $\LV(x^+)\le L_R$. \hfill$\square$

\textbf{Consequences.}
\begin{enumerate}[label=(\arabic*)]
\item The admissible window floor is
\begin{equation}
  \max\{R_c,R_{\min}^{\rm ceil}\}.
\end{equation}
It combines the theological gate and the geometric ceiling-floor: grace must expand the vessel sufficiently to absorb the converted resistance. For cleansed deaths, $\sigma^-\to0$, the ceiling-floor vanishes and $R_c$ alone remains binding.

\item The condition $\kappa_R>0$ suffices. A natural design choice is
\begin{equation}
  \kappa_R\ge\frac1{q^*},
\end{equation}
so that the asymptotic ceiling $q^+\to1/\kappa_R\le q^*$ and, in the intended window, $R_{\min}^{\rm ceil}\le R_c$.

\item The admissible grace window is
\begin{equation}
  \Grec^-\in
  \left[\max\{R_c,R_{\min}^{\rm ceil}\},R_{\max}\right].
\end{equation}
Below this window one obtains final death; above it one obtains over-dilution, $r<r_*$. The window is nonempty if and only if
\begin{equation}
  \Om^-(r^--r_*)
  \ge
  \kappa_R r_*\max\{R_c,R_{\min}^{\rm ceil}\}.
\end{equation}
A death pinned at $r^-=r_*$ is not resurrectible.

\item With the non-Zeno dwell time
\begin{equation}
  \Delta t_j\ge\frac{\bar h_*}{\bar M_*}>0,
\end{equation}
every life-arc satisfies
\begin{equation}
  \LV\le L_*:=
  \max\left\{L_R,\frac{C}{C_1}\right\},
\end{equation}
while admissible reset states themselves satisfy \(\LV(x^+)\le L_R\). Resets do not accumulate. Hence the hybrid trajectory is forward complete across any infinite sequence of life-arcs.
\end{enumerate}

\textbf{Theological gloss.} This remark is not used in the proof. The gate
\begin{equation}
  \Grec^-\ge R_c=C_\sigma-\sigma_0-\lambda_0
\end{equation}
is the threshold past which Sophia can no longer be extinguished by resistance; in that regime the balance forces $\lambda^-\ge0$, and strictly $\lambda^->0$ whenever $\sigma^-<C_\sigma$. Its complement, $\lambda^-\to0$, is the self-closing limit $\Igate\to0$: final impenitence. It resembles the oil of the wise virgins and the \emph{donum perseverantiae}. The upper boundary $R_{\max}$ represents the opposite failure: an over-diluting or ``cheap grace'' regime.

\section{Domain Stability Theorem (Complete Proof)}

This final section consolidates the preceding formulae into a single rigorous
statement: the active domain is positively invariant, forward complete, never
contacts the death-boundary, and exhibits asymptotic cleansing and alignment,
with admissible resets continuing the trajectory without Zeno accumulation. The
dissipative step is Proposition~\ref{prop:diss}; the light-filter constant is
Lemma~\ref{lem:cpsi}.

\subsection{Setup recap and hypotheses}
We use the normalized flow (Section~\ref{sec:witness} notation; $\Lambda_c=1$),
\begin{equation}\label{eq:flow2}
\begin{aligned}
  \dot h_1&=\mu(ah_2+cq)-(\rho+\alpha q)h_1, &
  \dot h_2&=\mu(ah_1+cq)-(\rho+\alpha q)h_2,\\
  \dot q&=\delta mT-\alpha q^2+\Sq, &
  \dot h_3&=\alpha q(1-h_3),\\
  \dot\sigma&=-\delta\,\Om\,m\,T, &
  \Sq&\in[0,\Sq^{\max}],
\end{aligned}
\end{equation}
with $\Om=1/(1-h_3)$, $T=\tanh(\kappa\sigma)$, $\mu=1-r$, $e=\varepsilon(1-h_3)^2$,
$r^2=h_1h_2+e$, $m=r+h_3-1$, and the Lyapunov functional
$\LV=A_\sigma\Psis+A_\Delta D_\Delta+A_qq^2+A_rB(r)$ with
$\Psis=\tfrac1\kappa\ln\cosh\kappa\sigma$, $D_\Delta=\tfrac12(h_1-h_2)^2$,
$B(r)=-\ln(1-r)$, weights $A_\bullet>0$.

\begin{definition}[Active domain]\label{def:DAfull}
\begin{equation*}
\DA=\left\{(\sigma,h_1,h_2,q,h_3):
\begin{array}{l}
\sigma\in[0,C_\sigma],\quad q\in[0,q^*],\quad h_3\in[h_{3,*},1),\quad r\in[r_*,r^*],\\
D_\Delta\le D^*,\quad h_1,h_2\ge0
\end{array}
\right\}.
\end{equation*}
Here $\mu_*=1-r^*$, $\Omega_{min}=1/(1-h_{3,*})$, $m_*=r_*+h_{3,*}-1$,
$e_{\max}=\varepsilon(1-h_{3,*})^2$, $d_*=\sqrt{2D^*}$, and
$q^*=1.05\sqrt{(\delta r^*+\Sq^{\max})/\alpha}$.
\end{definition}

\begin{assumption}[Parameters]\label{H:paramF}
$a,c,\alpha,\delta,\gamma,\zeta_0,\kappa,\varepsilon>0$, $\rho\ge0$.
\end{assumption}
\begin{assumption}[Walls]\label{H:wallsF}
$r_*^2>e_{\max}$; $m_*>0$; $\alpha(q^*)^2\ge\delta r^*+\Sq^{\max}$;
\begin{equation}
  (1-r^*)\!\left[a(d_*^2+2r^{*2})+cq^*\sqrt{d_*^2+4r^{*2}}\right]\le2\rho(r^{*2}-e_{\max}),
  \quad
  a(1-r_*)(r_*^2-e_{\max})\ge(\rho+\alpha q^*)r_*^2.
\end{equation}
\end{assumption}
\begin{assumption}[Reset window]\label{H:resetF}
At each death-event $\max\{R_c,R^{\mathrm{ceil}}_{\min}\}\le\Grec^-\le R_{\max}$,
$R_{\max}=\Om^-(r^--r_*)/(\kappa_R r_*)$, for the jump
$\sigma^+=q_R\sigma^-$, $\lambda^+=\lambda^-+(1-q_R)\sigma^-$,
$\Om^+=\Om^-+\kappa_R\Grec^-$, $(V,F)^+=(V,F)^-$, $q_R\in(0,1)$.
\end{assumption}

\subsection{Identities, regularity, invariance, completeness}

\begin{lemma}[Identities]\label{lem:identF}
Along \eqref{eq:flow2}, $2r\dot r=\Rrad:=\mu[a(h_1^2+h_2^2)+cq(h_1+h_2)]-2\rho(r^2-e)-2\alpha qr^2$
and $\dot D_\Delta=-2(a\mu+\rho+\alpha q)D_\Delta$.
\end{lemma}
\begin{proof}
Differentiate $r^2=h_1h_2+e(h_3)$ and use \eqref{eq:flow2}; the $h_1h_2$ and
$e$ contributions combine via $h_1h_2=r^2-e$ and $e'(h_3)\dot h_3=-2\alpha q\,e$.
For $D_\Delta$, $\frac{d}{dt}(h_1-h_2)=-(a\mu+\rho+\alpha q)(h_1-h_2)$.
\end{proof}

\begin{lemma}[Regularity]\label{lem:regF}
The field \eqref{eq:flow2} is $C^1$ near $\DA$ (since $VF+\varepsilon\ge\varepsilon>0$
and $h_3<1$), giving unique maximal solutions.
\end{lemma}

\begin{lemma}[Positive invariance --- whole-face Nagumo]\label{lem:nagumoF}
Under Hypotheses~\ref{H:paramF}--\ref{H:wallsF}, $\DA$ is positively invariant.
\end{lemma}
\begin{proof}
On $\DA$: $m\ge m_*>0$, $\mu\ge\mu_*$, $\Om\ge\Omega_{min}$, $T\in[0,1]$. We verify
sub-tangentiality face by face (Nagumo).
\emph{$\sigma{=}C_\sigma$:} $\dot\sigma=-\delta\Om m\tanh(\kappa C_\sigma)<0$.
\emph{$\sigma{=}0$:} $\dot\sigma=0$ (tangent).
\emph{$h_3{=}h_{3,*}$:} $\dot h_3=\alpha q(1-h_3)\ge0$ ($h_3$ non-decreasing).
\emph{$D_\Delta{=}D^*$:} $\dot D_\Delta\le0$ by Lemma~\ref{lem:identF}.
\emph{$q{=}0$:} $\dot q=\delta mT+\Sq\ge0$ (no lower $q$-floor needed).
\emph{$q{=}q^*$:} $\dot q\le\delta r^*+\Sq^{\max}-\alpha(q^*)^2\le0$ by the $q$-wall.
\emph{$r{=}r^*$:} writing $\Rrad=P-N$ with $P=\mu_*[a(h_1^2+h_2^2)+cq(h_1+h_2)]$,
$N=2\rho(r^{*2}-e)+2\alpha qr^{*2}$, the face bounds $(h_1-h_2)^2\le d_*^2$,
$0\le e\le e_{\max}$, $0\le q\le q^*$ give
$\sup P\le\mu_*[a(d_*^2+2r^{*2})+cq^*\sqrt{d_*^2+4r^{*2}}]$ and
$\inf N\ge2\rho(r^{*2}-e_{\max})$, so $\sup\Rrad\le\sup P-\inf N\le0$ by the upper
radial wall; hence $\dot r\le0$.
\emph{$r{=}r_*$:} symmetrically $\inf P\ge2a(1-r_*)(r_*^2-e_{\max})$,
$\sup N\le2(\rho+\alpha q^*)r_*^2$, so $\inf\Rrad\ge0$ by the lower radial wall;
hence $\dot r\ge0$. All faces sub-tangential $\Rightarrow\DA$ invariant.
\end{proof}

\begin{lemma}[Forward completeness]\label{lem:completeF}
Every solution from $\DA$ exists on $[0,\infty)$, with
$1-h_3(t)\ge(1-h_3(0))e^{-\alpha q^*t}>0$, so $h_3<1$ and $\Om<\infty$ on finite
intervals.
\end{lemma}
\begin{proof}
By Lemma~\ref{lem:nagumoF} the orbit stays in $\DA$ (all coordinates but $\Om$
bounded); from $\dot h_3=\alpha q(1-h_3)$, $q\le q^*$, integrate
$\frac{d}{dt}(1-h_3)\ge-\alpha q^*(1-h_3)$. No finite-time blow-up, so the orbit
extends to $[0,\infty)$.
\end{proof}

\subsection{Dissipation, alignment, cleansing, barrier}

The dissipative estimate $\dot\LV\le C-C_1\LV$ with
$C_1=\min\{\delta\Omega_{min}m_*c_\Psi,\,2a\mu_*\}>0$ is
Proposition~\ref{prop:diss} (using Lemma~\ref{lem:cpsi}).

\begin{lemma}[Alignment]\label{lem:alignF}
$D_\Delta(t)\le D_\Delta(0)e^{-2a\mu_*t}\to0$.
\end{lemma}
\begin{proof}
$\dot D_\Delta=-2(a\mu+\rho+\alpha q)D_\Delta\le-2a\mu_*D_\Delta$; integrate.
\end{proof}

\begin{lemma}[Cleansing]\label{lem:cleanF}
$\sigma(t)\to0$ as $t\to\infty$.
\end{lemma}
\begin{proof}
On $\DA$, $\dot\sigma=-\delta\Om m\tanh(\kappa\sigma)\le-\beta\tanh(\kappa\sigma)$,
$\beta=\delta\Omega_{min}m_*>0$. Thus $\sigma$ is nonincreasing, bounded below by
$0$, hence $\sigma\to\sigma_\infty\ge0$; comparison with
$\dot y=-\beta\tanh(\kappa y)$ (whose only, globally attracting, equilibrium is
$0$) forces $\sigma_\infty=0$.
\end{proof}

\begin{corollary}[Death-boundary never contacted]\label{cor:barrierF}
For all $t\ge0$,
\begin{equation*}
  r(t)\le r^*<1,\qquad
  \LV(t)\le\max\{\LV(0),C/C_1\}.
\end{equation*}
Consequently $u=r\Om<\Om$ always.
\end{corollary}

\subsection{Hybrid layer}

\begin{lemma}[Reset admissibility]\label{lem:resetF}
Under Hypothesis~\ref{H:resetF}, $x^-\in\DA\Rightarrow\reset(x^-)\in\DA$, with
$r^+\le r^-$, $\sigma^+\le\sigma^-$, $D_\Delta^+\le D_\Delta^-$, $h_3^+\ge h_3^-$.
\end{lemma}
\begin{proof}
$(V,F)$ preserved $\Rightarrow u^+=u^-$, $\Om^+=\Om^-+\kappa_R\Grec^-\ge\Om^-$, so
$r^+=r^-\Om^-/\Om^+\le r^-$, $h_3^+\ge h_3^-$, $\sigma^+=q_R\sigma^-\le\sigma^-$,
$D_\Delta^+=(\Om^-/\Om^+)^2D_\Delta^-\le D_\Delta^-$. The window upper bound
$\Grec^-\le R_{\max}$ is equivalent to $r^+\ge r_*$. Since $x^-\in\DA$ gives $\lambda^-\ge0$, one has $q^+\ge0$; and the ceiling-floor condition gives $q^+\le q^*$. Thus $q^+\in[0,q^*]$. All constraints of $\DA$ hold; the margin
$m^+\ge m_*$ follows from $r^+\ge r_*$, $h_3^+\ge h_{3,*}$.
\end{proof}

\begin{lemma}[Non-Zeno]\label{lem:nonzenoF}
The field is bounded on $\DA$ ($\sup\|f\|=\bar M_*<\infty$) and resets land with a
uniform interior margin $\bar h_*>0$; hence consecutive resets are separated by
$\Delta t_j\ge\bar h_*/\bar M_*>0$, so the hybrid trajectory is non-Zeno and
forward complete.
\end{lemma}

\subsection{Main theorem}

\begin{theorem}[Domain stability of $\DA$]\label{thm:mainF}
Assume Hypotheses~\ref{H:paramF}--\ref{H:resetF}. Then for every $x(0)\in\DA$:
\textnormal{(i)} $x(t)\in\DA\ \forall t\ge0$;
\textnormal{(ii)} the solution exists on $[0,\infty)$ with $h_3(t)<1$,
$\Om(t)<\infty$ on finite intervals;
\textnormal{(iii)} $\LV(t)\le\max\{\LV(0),C/C_1\}$, $C_1>0$, and $r(t)\le r^*<1$
(death-boundary never contacted);
\textnormal{(iv)} $\sigma(t)\to0$ and $D_\Delta(t)\to0$;
\textnormal{(v)} under admissible resets the hybrid trajectory stays in $\DA$, is
non-Zeno, and forward complete. Schematically,
\begin{equation}
\boxed{
\text{Hyp.}~\ref{H:paramF}\text{--}\ref{H:wallsF}
\Rightarrow\DA\ \text{invariant}
\Rightarrow\text{complete}
\Rightarrow\dot\LV\le C-C_1\LV
\Rightarrow r<1
\Rightarrow \sigma,D_\Delta\to0.}
\end{equation}
\end{theorem}
\begin{proof}
(i) Lemma~\ref{lem:nagumoF}; (ii) Lemma~\ref{lem:completeF}; (iii)
Proposition~\ref{prop:diss} with Corollary~\ref{cor:barrierF}; (iv)
Lemmas~\ref{lem:alignF},~\ref{lem:cleanF}, using the invariant lower bounds
$\mu\ge\mu_*$, $m\ge m_*$ from (i); (v) Lemmas~\ref{lem:resetF},~\ref{lem:nonzenoF}.
\end{proof}

\begin{remark}[Nature of the stability]
This is \emph{set} stability with \emph{partial} convergence: $\DA$ is invariant
and $(\sigma,D_\Delta)\to0$, while $(q,\Om)$ need not converge ---
$\dot\Om=\alpha\lambda>0$ for $\lambda>0$, so $\Om\to\infty$ under sustained
inflow (\emph{epektasis}). There is no equilibrium; $\LV$ is a
boundedness-and-barrier certificate, not a function vanishing at a rest point.
The conclusion is strictly domain-conditional ($x(0)\in\DA$), not global or
automatic.
\end{remark}

\begin{remark}[Closed limit]
At $\zeta_0=0$: $\Sq=\Igate=0$, $\dot\lambda=-\dot\sigma$, and
$\sigma+\lambda=\sigma_0+\lambda_0$ (exact conservation). The Open-Gate model adds
only the bounded $\Sq$ in $\dot q$: it strengthens the walls (a larger $q^*$) but
leaves the dissipative core unchanged.
\end{remark}

\begin{remark}[Grace enters $C$, never $C_1$]
$\Igate$ enters \eqref{eq:flow2} only through $\Sq$, feeding the bounded blocks of
Proposition~\ref{prop:diss}; it never enters $\kappa_\sigma$ or $\kappa_\Delta$.
Grace raises $C$ by at most $\Delta C_{\mathrm{gate}}\le2A_qq^*\Sq^{\max}$ and can
neither manufacture nor destroy $C_1>0$: the transforming structure is sustained,
not replaced. Moreover $\inf C_1=0$ at the admissibility boundary
($m_*\to0^+$ or $\mu_*\to0^+$): a state pinned on the active margin has zero
cleansing rate --- stability holds, transfiguration stalls.
\end{remark}
